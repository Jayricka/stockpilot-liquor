import {
  ChevronDown,
  Package,
  Plus,
} from 'lucide-react'

function ProductHeader({
  businesses,
  businessId,
  onBusinessChange,
  productCount,
  onAddProduct,
}) {
  return (
    <section className="products-header-card">
      <div className="products-business">
        <div className="products-business-icon">
          <Package size={20} />
        </div>

        <div>
          <span className="products-eyebrow">
            Business
          </span>

          <div className="products-business-select">
            <select
              value={businessId || ''}
              onChange={(event) =>
                onBusinessChange(
                  event.target.value,
                )
              }
            >
              {businesses.map((business) => (
                <option
                  key={business.id}
                  value={business.id}
                >
                  {business.name}
                </option>
              ))}
            </select>

            <ChevronDown size={16} />
          </div>

          <span className="products-count">
            {productCount} product
            {productCount === 1 ? '' : 's'}
          </span>
        </div>
      </div>

      <button
        type="button"
        className="products-add-button"
        onClick={onAddProduct}
      >
        <Plus size={17} />
        Add product
      </button>
    </section>
  )
}

export default ProductHeader
import {
  Filter,
  Search,
  X,
} from 'lucide-react'

const GROUP_OPTIONS = [
  {
    value: 'all',
    label: 'All products',
  },
  {
    value: 'ALCOHOLIC_DRINKS',
    label: 'Alcoholic Drinks',
  },
  {
    value: 'SODA_AND_DRINKS',
    label: 'Soda & Drinks',
  },
  {
    value: 'CIGARETTES_AND_TOBACCO',
    label: 'Cigarettes & Tobacco',
  },
  {
    value: 'OTHER',
    label: 'Other',
  },
]

function ProductFilters({
  search,
  groupFilter,
  categoryFilter,
  brandFilter,
  stockFilter,
  statusFilter,
  categories,
  brands,
  onSearchChange,
  onGroupChange,
  onCategoryChange,
  onBrandChange,
  onStockChange,
  onStatusChange,
  resultCount,
}) {
  const hasFilters =
    Boolean(search) ||
    groupFilter !== 'all' ||
    categoryFilter !== 'all' ||
    brandFilter !== 'all' ||
    stockFilter !== 'all' ||
    statusFilter !== 'all'

  function clearFilters() {
    onSearchChange('')
    onGroupChange('all')
    onCategoryChange('all')
    onBrandChange('all')
    onStockChange('all')
    onStatusChange('all')
  }

  return (
    <section className="products-filters-card">
      <div className="products-group-tabs">
        {GROUP_OPTIONS.map((group) => (
          <button
            key={group.value}
            type="button"
            className={
              groupFilter === group.value
                ? 'active'
                : ''
            }
            onClick={() =>
              onGroupChange(group.value)
            }
          >
            {group.label}
          </button>
        ))}
      </div>

      <div className="products-filter-row">
        <div className="products-search">
          <Search size={18} />

          <input
            type="search"
            placeholder="Search name, brand, SKU or category..."
            value={search}
            onChange={(event) =>
              onSearchChange(
                event.target.value,
              )
            }
          />
        </div>

        <div className="products-filter-select">
          <Filter size={16} />

          <select
            value={categoryFilter || 'all'}
            onChange={(event) =>
              onCategoryChange(
                event.target.value,
              )
            }
          >
            <option value="all">
              All categories
            </option>

            {categories.map((category) => (
              <option
                key={category.id}
                value={String(category.id)}
              >
                {category.name}
              </option>
            ))}
          </select>
        </div>

        <div className="products-filter-select">
          <select
            value={brandFilter || 'all'}
            onChange={(event) =>
              onBrandChange(
                event.target.value,
              )
            }
          >
            <option value="all">
              All brands
            </option>

            {brands.map((brand) => (
              <option
                key={brand}
                value={brand}
              >
                {brand}
              </option>
            ))}
          </select>
        </div>

        <div className="products-filter-select">
          <select
            value={stockFilter || 'all'}
            onChange={(event) =>
              onStockChange(
                event.target.value,
              )
            }
          >
            <option value="all">
              All stock
            </option>

            <option value="in-stock">
              In stock
            </option>

            <option value="low-stock">
              Low stock
            </option>

            <option value="out-of-stock">
              Out of stock
            </option>
          </select>
        </div>

        <div className="products-filter-select">
          <select
            value={statusFilter || 'all'}
            onChange={(event) =>
              onStatusChange(
                event.target.value,
              )
            }
          >
            <option value="all">
              All status
            </option>

            <option value="active">
              Active
            </option>

            <option value="inactive">
              Inactive
            </option>
          </select>
        </div>

        {hasFilters && (
          <button
            type="button"
            className="products-clear-filter"
            onClick={clearFilters}
          >
            <X size={15} />
            Clear
          </button>
        )}

        <span className="products-result-count">
          {resultCount}{' '}
          {resultCount === 1
            ? 'result'
            : 'results'}
        </span>
      </div>
    </section>
  )
}

export default ProductFilters
import {
  Edit3,
  PackagePlus,
  Trash2,
} from 'lucide-react'

function ProductTable({
  products,
  loading,
  onEdit,
  onDelete,
  onAddProduct,
}) {
  return (
    <section className="products-table-card">
      <header className="products-table-heading">
        <div>
          <span>Product catalogue</span>
          <h2>Products</h2>
        </div>

        <span>
          {products.length} product
          {products.length === 1 ? '' : 's'}
        </span>
      </header>

      {loading ? (
        <div className="products-table-loading">
          Loading products...
        </div>
      ) : products.length === 0 ? (
        <div className="products-empty">
          <PackagePlus size={28} />

          <h3>No products found</h3>

          <p>
            Add a product or adjust your filters.
          </p>

          <button
            type="button"
            onClick={onAddProduct}
          >
            <PackagePlus size={16} />
            Add product
          </button>
        </div>
      ) : (
        <div className="products-table-wrapper">
          <table className="products-table">
            <thead>
              <tr>
                <th>Product</th>
                <th>Brand</th>
                <th>Category</th>
                <th>SKU</th>
                <th>Unit</th>
                <th>Stock</th>
                <th>Buying</th>
                <th>Selling</th>
                <th>Status</th>
                <th>Actions</th>
              </tr>
            </thead>

            <tbody>
              {products.map((product) => (
                <tr key={product.id}>
                  <td>
                    <strong>{product.name}</strong>
                  </td>

                  <td>
                    {product.brand || '—'}
                  </td>

                  <td>
                    <div className="product-category-cell">
                      <strong>
                        {product.category_name || '—'}
                      </strong>

                      {product.category_group_display && (
                        <span>
                          {
                            product.category_group_display
                          }
                        </span>
                      )}
                    </div>
                  </td>

                  <td>{product.sku || '—'}</td>

                  <td>{product.unit}</td>

                  <td>
                    <strong>
                      {product.stock_quantity}
                    </strong>
                  </td>

                  <td>
                    Ksh{' '}
                    {Number(
                      product.buying_price || 0,
                    ).toLocaleString()}
                  </td>

                  <td>
                    Ksh{' '}
                    {Number(
                      product.selling_price || 0,
                    ).toLocaleString()}
                  </td>

                  <td>
                    <span
                      className={`products-status ${
                        product.is_low_stock
                          ? 'status-warning'
                          : product.is_active
                            ? 'status-success'
                            : 'status-danger'
                      }`}
                    >
                      {product.is_low_stock
                        ? 'Low stock'
                        : product.is_active
                          ? 'Active'
                          : 'Inactive'}
                    </span>
                  </td>

                  <td>
                    <div className="products-actions">
                      <button
                        type="button"
                        title="Edit product"
                        aria-label={`Edit ${product.name}`}
                        onClick={() =>
                          onEdit(product)
                        }
                      >
                        <Edit3 size={15} />
                      </button>

                      <button
                        type="button"
                        title="Delete product"
                        aria-label={`Delete ${product.name}`}
                        onClick={() =>
                          onDelete(product)
                        }
                      >
                        <Trash2 size={15} />
                      </button>
                    </div>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </section>
  )
}

export default ProductTable
import { useMemo, useState } from 'react'
import { X } from 'lucide-react'

const EMPTY_FORM = {
  category: '',
  brand: '',
  name: '',
  sku: '',
  unit: 'BOTTLE',
  initial_quantity: '0',
  buying_price: '',
  selling_price: '',
  reorder_level: '5',
  is_active: true,
}

const UNIT_OPTIONS = [
  {
    value: 'BOTTLE',
    label: 'Bottle',
  },
  {
    value: 'LITRE',
    label: 'Litre',
  },
  {
    value: 'CRATE',
    label: 'Crate',
  },
  {
    value: 'PIECE',
    label: 'Piece',
  },
]

function getInitialForm(product) {
  if (!product) {
    return { ...EMPTY_FORM }
  }

  return {
    category:
      product.category?.id
        ? String(product.category.id)
        : product.category
          ? String(product.category)
          : product.category_id
            ? String(product.category_id)
            : '',
    brand:
      typeof product.brand === 'string'
        ? product.brand
        : product.brand?.name || '',
    name: product.name || '',
    sku: product.sku || '',
    unit: product.unit || 'BOTTLE',
    initial_quantity: '0',
    buying_price:
      product.buying_price ?? '',
    selling_price:
      product.selling_price ?? '',
    reorder_level:
      product.reorder_level ?? '5',
    is_active:
      product.is_active ?? true,
  }
}

function ProductForm({
  product,
  categories,
  loading,
  errors,
  onSubmit,
  onClose,
}) {
  const [form, setForm] = useState(() =>
    getInitialForm(product),
  )

  const groupedCategories = useMemo(() => {
    return categories.reduce(
      (groups, category) => {
        const groupKey =
          category.group ||
          category.category_group ||
          'OTHER'

        const groupLabel =
          category.group_display ||
          category.category_group_display ||
          'Other'

        if (!groups[groupKey]) {
          groups[groupKey] = {
            value: groupKey,
            label: groupLabel,
            categories: [],
          }
        }

        groups[groupKey].categories.push(
          category,
        )

        return groups
      },
      {},
    )
  }, [categories])

  function handleChange(event) {
    const {
      name,
      value,
      type,
      checked,
    } = event.target

    setForm((currentForm) => ({
      ...currentForm,
      [name]:
        type === 'checkbox'
          ? checked
          : value,
    }))
  }

  function getFieldError(fieldName) {
    const fieldError =
      errors?.[fieldName]

    if (Array.isArray(fieldError)) {
      return fieldError.join(' ')
    }

    if (
      fieldError &&
      typeof fieldError === 'object'
    ) {
      return Object.values(fieldError)
        .flat()
        .join(' ')
    }

    return fieldError || ''
  }

  function handleSubmit(event) {
    event.preventDefault()

    const categoryId = Number(
      form.category,
    )

    const buyingPrice = Number(
      form.buying_price,
    )

    const sellingPrice = Number(
      form.selling_price,
    )

    const reorderLevel = Number(
      form.reorder_level,
    )

    const initialQuantity = Number(
      form.initial_quantity,
    )

    const payload = {
      category: categoryId,
      name: form.name.trim(),
      unit: form.unit,
      buying_price: buyingPrice,
      selling_price: sellingPrice,
      reorder_level: reorderLevel,
      is_active: Boolean(
        form.is_active,
      ),
    }

    const brand = form.brand.trim()

    if (brand) {
      payload.brand = brand
    }

    const sku = form.sku.trim()

    if (sku) {
      payload.sku = sku
    }

    if (!product) {
      payload.initial_quantity =
        initialQuantity
    }

    onSubmit(payload)
  }

  const generalError =
    errors?.detail ||
    errors?.non_field_errors

  return (
    <div
      className="product-modal-backdrop"
      role="presentation"
      onMouseDown={(event) => {
        if (
          event.target ===
          event.currentTarget
        ) {
          onClose()
        }
      }}
    >
      <section
        className="product-modal"
        role="dialog"
        aria-modal="true"
        aria-labelledby="product-form-title"
      >
        <header className="product-modal-header">
          <div>
            <span className="product-modal-eyebrow">
              Product catalogue
            </span>

            <h2 id="product-form-title">
              {product
                ? 'Edit product'
                : 'Add product'}
            </h2>
          </div>

          <button
            type="button"
            className="product-modal-close"
            aria-label="Close product form"
            onClick={onClose}
            disabled={loading}
          >
            <X size={20} />
          </button>
        </header>

        {generalError && (
          <div className="product-form-error">
            {Array.isArray(generalError)
              ? generalError.join(' ')
              : generalError}
          </div>
        )}

        <form
          className="product-form"
          onSubmit={handleSubmit}
        >
          <section className="product-form-section">
            <div className="product-form-section-heading">
              <h3>Product details</h3>

              <p>
                Identify the product and
                its category.
              </p>
            </div>

            <div className="product-form-grid">
              <div className="product-form-field">
                <label htmlFor="category">
                  Category
                </label>

                <select
                  id="category"
                  name="category"
                  value={form.category}
                  onChange={handleChange}
                  required
                >
                  <option value="">
                    Select category
                  </option>

                  {Object.values(
                    groupedCategories,
                  ).map((group) => (
                    <optgroup
                      key={group.value}
                      label={group.label}
                    >
                      {group.categories.map(
                        (category) => (
                          <option
                            key={category.id}
                            value={String(
                              category.id,
                            )}
                          >
                            {category.name}
                          </option>
                        ),
                      )}
                    </optgroup>
                  ))}
                </select>

                {getFieldError(
                  'category',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'category',
                    )}
                  </span>
                )}
              </div>

              <div className="product-form-field">
                <label htmlFor="brand">
                  Brand
                </label>

                <input
                  id="brand"
                  name="brand"
                  type="text"
                  value={form.brand}
                  placeholder="e.g. Gordon's"
                  onChange={handleChange}
                />

                {getFieldError(
                  'brand',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'brand',
                    )}
                  </span>
                )}
              </div>

              <div className="product-form-field">
                <label htmlFor="name">
                  Product name
                </label>

                <input
                  id="name"
                  name="name"
                  type="text"
                  value={form.name}
                  placeholder="e.g. London Dry Gin"
                  onChange={handleChange}
                  required
                />

                {getFieldError(
                  'name',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'name',
                    )}
                  </span>
                )}
              </div>

              <div className="product-form-field">
                <label htmlFor="sku">
                  SKU
                </label>

                <input
                  id="sku"
                  name="sku"
                  type="text"
                  value={form.sku}
                  placeholder="e.g. GIN-001"
                  onChange={handleChange}
                />

                {getFieldError(
                  'sku',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'sku',
                    )}
                  </span>
                )}
              </div>
            </div>
          </section>

          <section className="product-form-section">
            <div className="product-form-section-heading">
              <h3>Pricing and stock</h3>

              <p>
                Set prices and inventory
                thresholds.
              </p>
            </div>

            <div className="product-form-grid">
              <div className="product-form-field">
                <label htmlFor="unit">
                  Unit
                </label>

                <select
                  id="unit"
                  name="unit"
                  value={form.unit}
                  onChange={handleChange}
                  required
                >
                  {UNIT_OPTIONS.map(
                    (unit) => (
                      <option
                        key={unit.value}
                        value={unit.value}
                      >
                        {unit.label}
                      </option>
                    ),
                  )}
                </select>

                {getFieldError(
                  'unit',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'unit',
                    )}
                  </span>
                )}
              </div>

              {!product && (
                <div className="product-form-field">
                  <label htmlFor="initial_quantity">
                    Initial quantity
                  </label>

                  <input
                    id="initial_quantity"
                    name="initial_quantity"
                    type="number"
                    min="0"
                    step="1"
                    value={
                      form.initial_quantity
                    }
                    onChange={handleChange}
                  />

                  {getFieldError(
                    'initial_quantity',
                  ) && (
                    <span className="product-field-error">
                      {getFieldError(
                        'initial_quantity',
                      )}
                    </span>
                  )}
                </div>
              )}

              <div className="product-form-field">
                <label htmlFor="buying_price">
                  Buying price
                </label>

                <input
                  id="buying_price"
                  name="buying_price"
                  type="number"
                  min="0"
                  step="0.01"
                  value={
                    form.buying_price
                  }
                  placeholder="0.00"
                  onChange={handleChange}
                  required
                />

                {getFieldError(
                  'buying_price',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'buying_price',
                    )}
                  </span>
                )}
              </div>

              <div className="product-form-field">
                <label htmlFor="selling_price">
                  Selling price
                </label>

                <input
                  id="selling_price"
                  name="selling_price"
                  type="number"
                  min="0"
                  step="0.01"
                  value={
                    form.selling_price
                  }
                  placeholder="0.00"
                  onChange={handleChange}
                  required
                />

                {getFieldError(
                  'selling_price',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'selling_price',
                    )}
                  </span>
                )}
              </div>

              <div className="product-form-field">
                <label htmlFor="reorder_level">
                  Reorder level
                </label>

                <input
                  id="reorder_level"
                  name="reorder_level"
                  type="number"
                  min="0"
                  step="1"
                  value={
                    form.reorder_level
                  }
                  onChange={handleChange}
                  required
                />

                {getFieldError(
                  'reorder_level',
                ) && (
                  <span className="product-field-error">
                    {getFieldError(
                      'reorder_level',
                    )}
                  </span>
                )}
              </div>
            </div>
          </section>

          <section className="product-form-section">
            <label className="product-form-checkbox">
              <input
                type="checkbox"
                name="is_active"
                checked={Boolean(
                  form.is_active,
                )}
                onChange={handleChange}
              />

              <span>
                <strong>
                  Active product
                </strong>

                <small>
                  Allow this product to
                  appear in active
                  catalogue views.
                </small>
              </span>
            </label>
          </section>

          <footer className="product-form-actions">
            <button
              type="button"
              className="product-secondary-button"
              onClick={onClose}
              disabled={loading}
            >
              Cancel
            </button>

            <button
              type="submit"
              className="product-primary-button"
              disabled={loading}
            >
              {loading
                ? 'Saving...'
                : product
                  ? 'Update product'
                  : 'Create product'}
            </button>
          </footer>
        </form>
      </section>
    </div>
  )
}

export default ProductForm
import {
  Boxes,
  CircleCheck,
  AlertTriangle,
  Wallet,
} from 'lucide-react'

function ProductStats({ products }) {
  const total = products.length

  const active = products.filter(
    (product) => product.is_active,
  ).length

  const lowStock = products.filter(
    (product) => product.is_low_stock,
  ).length

  const inventoryValue = products.reduce(
    (totalValue, product) =>
      totalValue +
      Number(product.buying_price || 0) *
        Number(product.stock_quantity || 0),
    0,
  )

  const stats = [
    {
      label: 'Total products',
      value: total,
      icon: Boxes,
    },
    {
      label: 'Active products',
      value: active,
      icon: CircleCheck,
    },
    {
      label: 'Low stock',
      value: lowStock,
      icon: AlertTriangle,
    },
    {
      label: 'Inventory cost value',
      value: `Ksh ${inventoryValue.toLocaleString()}`,
      icon: Wallet,
    },
  ]

  return (
    <div className="products-stats">
      {stats.map((stat) => {
        const Icon = stat.icon

        return (
          <article
            key={stat.label}
            className="products-stat"
          >
            <div className="products-stat-icon">
              <Icon size={17} />
            </div>

            <span>{stat.label}</span>

            <strong>{stat.value}</strong>
          </article>
        )
      })}
    </div>
  )
}

export default ProductStats
